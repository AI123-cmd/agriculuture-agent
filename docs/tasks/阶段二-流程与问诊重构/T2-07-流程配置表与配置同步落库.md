# T2-07 流程配置表与配置同步落库

## 这一项要做什么

在 `app/models/process_config.py` 与 `app/repositories/process_config.py` 中建立流程配置与流程红线的持久化结构，并实现"从 seeds 到数据库"的同步任务。数据库成为配置的运行时事实来源，seeds 只在首次部署时作为初始数据源。

落库的目的有三个：一是让热加载有比对目标（T2-05 从这里读最新版本），二是让历史问诊能追溯当时生效的配置版本，三是让后台修订配置不必重新发版。所有查询走仓储层，不使用 `db.query(Model)`，符合全局约定第四节的 L4 规则。

## 前置依赖

- 依赖 T2-02：必须已提供 `ProcessConfig` 及其子模型，落库时序列化为 `config_json`
- 依赖 T2-03：必须已提供加载器，同步任务的输入是加载结果
- 依赖 T2-06：必须已提供 `LoadedRedlines`，红线单独落表便于按规则 id 反查
- 依赖 T1-04：必须已建立 Alembic 迁移链路，本项新增迁移 `0002_process_config.py`
- 外部环境：PostgreSQL 16+，`process_config` 与 `process_redline` 两张表

## 实现要点

### 流程配置表

```python

class ProcessConfigRow(Base):
    """流程配置表。版本化：每次修改插入新行，不覆盖历史。"""
    __tablename__ = "process_config"
    id = mapped_column(String(36), primary_key=True)
    process_id = mapped_column(String(20), nullable=False, index=True)
    config_version = mapped_column(Integer, nullable=False)
    display_name = mapped_column(String(50), nullable=False)
    status = mapped_column(String(20), nullable=False, default="approved")  # draft/approved/deprecated
    config_json = mapped_column(JSONB, nullable=False)        # ProcessConfig 的规范化序列化
    content_hash = mapped_column(String(64), nullable=False)  # 内容指纹，供热加载比对
    source = mapped_column(String(20), nullable=False, default="seed")  # seed/manual
    note = mapped_column(Text, nullable=True)                 # 修订说明
    effective_from = mapped_column(DateTime(timezone=True), nullable=False, server_default=func.now())
    effective_to = mapped_column(DateTime(timezone=True), nullable=True)  # 为空表示当前生效
    created_at = mapped_column(DateTime(timezone=True), nullable=False, server_default=func.now())
    __table_args__ = (
        UniqueConstraint("process_id", "config_version", name="uq_process_config_version"),
        Index("ix_process_config_active", "process_id", "effective_to"),
    )
```

配置表不带 `user_id`，因为它是全局配置不是农户数据。这与全局约定"所有业务表带 `user_id`"不冲突：该约定的目的是防越权，而配置表对所有农户公开且只读，没有归属概念。这一点要在表注释中写明，避免后续被当作遗漏补上。

`effective_to` 的设计让"当前生效版本"可以一次查询拿到（`WHERE effective_to IS NULL`），也保留了完整时间轴。不使用 `is_active` 布尔，是因为布尔无法回答"这条配置从什么时候开始生效"，而审计表需要与时间对齐。

### 流程红线表

```python

class ProcessRedlineRow(Base):
    """流程红线表。与配置表同版本，独立成表以便按规则 id 反查。"""
    __tablename__ = "process_redline"
    id = mapped_column(String(36), primary_key=True)
    process_id = mapped_column(String(20), nullable=False, index=True)
    config_version = mapped_column(Integer, nullable=False)
    redline_id = mapped_column(String(30), nullable=False)     # 如 TEA-01
    desc = mapped_column(Text, nullable=False)
    rule_ids = mapped_column(JSONB, nullable=False)            # ["R-INTERVAL", "R-MAX-USE"]
    severity = mapped_column(String(10), nullable=False)       # block / warn
    payload = mapped_column(JSONB, nullable=False, default=dict)
    created_at = mapped_column(DateTime(timezone=True), nullable=False, server_default=func.now())
    __table_args__ = (
        UniqueConstraint("process_id", "config_version", "redline_id", name="uq_process_redline"),
        Index("ix_process_redline_rule", "rule_ids", postgresql_using="gin"),
    )
```

红线独立成表的价值在 `ix_process_redline_rule` 这个 GIN 索引上：闸门复盘时需要回答"所有引用了 `R-INTERVAL` 的流程有哪些"，这在单表 JSON 里很难查，独立成表加索引后是一次简单查询。红线表同时是"声明了但没执行"反查的依据。

### 仓储层

```python

class ProcessConfigRepository:
    """流程配置仓储。所有查询经此，不使用 db.query(Model)。"""
    def __init__(self, session: AsyncSession) -> None:
        self._session = session
    async def latest(self, process_id: ProcessId) -> ProcessConfigRow | None:
        stmt = (
            select(ProcessConfigRow)
            .where(ProcessConfigRow.process_id == process_id.value)
            .where(ProcessConfigRow.effective_to.is_(None))
            .order_by(ProcessConfigRow.config_version.desc())
            .limit(1)
        )
        return (await self._session.execute(stmt)).scalar_one_or_none()
    async def latest_per_process(self) -> dict[ProcessId, ProcessConfigRow]:
        stmt = (
            select(ProcessConfigRow)
            .where(ProcessConfigRow.effective_to.is_(None))
            .order_by(ProcessConfigRow.process_id, ProcessConfigRow.config_version.desc())
            .distinct(ProcessConfigRow.process_id)
        )
        rows = (await self._session.execute(stmt)).scalars().all()
        return {ProcessId(r.process_id): r for r in rows}
    async def by_version(self, process_id: ProcessId, version: int) -> ProcessConfigRow | None:
        stmt = select(ProcessConfigRow).where(
            ProcessConfigRow.process_id == process_id.value,
            ProcessConfigRow.config_version == version,
        )
        return (await self._session.execute(stmt)).scalar_one_or_none()
    async def insert_version(self, config: ProcessConfig, *, source: str, note: str | None) -> None:
        """插入新版本，并把上一版本的 effective_to 补齐。同事务内完成。"""
        now = datetime.now(timezone.utc)
        await self._session.execute(
            update(ProcessConfigRow)
            .where(ProcessConfigRow.process_id == config.process_id.value)
            .where(ProcessConfigRow.effective_to.is_(None))
            .values(effective_to=now)
        )
        self._session.add(ProcessConfigRow(
            id=str(uuid4()),
            process_id=config.process_id.value,
            config_version=config.version,
            display_name=config.display_name,
            config_json=config.model_dump(mode="json"),
            content_hash=compute_config_hash(config),
            source=source,
            note=note,
            effective_from=now,
        ))
```

`insert_version` 的关键是"先封旧再插新"在同一事务里。若分成两次提交，中间时刻会出现两个 `effective_to IS NULL` 的版本，`latest` 查询就会不确定。同一个时间戳写 `effective_to` 与 `effective_from`，让版本区间首尾相接。

`by_version` 是历史问诊复现的入口：给定 `consultation.config_version` 就能取出当时的完整配置。

### 同步任务

```python

class ConfigSyncService:
    """把 seeds 或人工提交的配置同步到数据库。幂等：内容未变则不写新版本。"""
    def __init__(self, session: AsyncSession) -> None:
        self._repo = ProcessConfigRepository(session)
    async def sync_from_seeds(self, *, dry_run: bool = False) -> list[SyncReport]:
        reports: list[SyncReport] = []
        for pid, config in load_all_configs().items():
            current = await self._repo.latest(pid)
            incoming_hash = compute_config_hash(config)
            if current is not None and current.content_hash == incoming_hash:
                reports.append(SyncReport(pid, action="skip", reason="内容未变"))
                continue
            next_version = (current.config_version + 1) if current else config.version
            bumped = config.model_copy(update={"version": next_version})
            if dry_run:
                reports.append(SyncReport(pid, action="would_insert", version=next_version))
                continue
            await self._repo.insert_version(bumped, source="seed", note="seeds 同步")
            await self._sync_redlines(bumped)
            reports.append(SyncReport(pid, action="inserted", version=next_version))
        return reports
```

幂等通过 `content_hash` 比对实现，而不是比对 `version`。文件里的 `version` 是编写者手写的，容易忘记递增；数据库里的 `config_version` 由同步任务自增，是真正的版本序列。这样"更新配置"这个动作只需改 YAML，版本号由系统维护。

`dry_run` 是必要的：在 CI 与预发环境先跑一次 dry-run 看会插入几条，避免误改一次配置就在生产插入一堆版本。

### 迁移要点

迁移中要显式写出两张表、唯一约束与索引，并在 `downgrade` 里 drop。`config_json` 与 `payload` 用 `JSONB` 而非 `JSON`，因为需要在其上做路径查询与 GIN 索引。`rule_ids` 用 `JSONB` 存数组，配套 `postgresql_using="gin"` 索引。

迁移不回填历史数据，首次同步由 `sync_from_seeds` 完成。种子同步任务在应用启动时以 ARQ 任务方式触发一次，或由部署脚本显式调用，二者都不得放在请求路径上。

## 验收标准

- `alembic upgrade head` 与 `alembic downgrade -1` 均返回 0，`alembic current` 显示 `0002_process_config`
- 数据库中存在表 `process_config` 与 `process_redline`，`\d process_config` 可见唯一约束 `uq_process_config_version` 与索引 `ix_process_config_active`
- `pytest tests/repositories/test_process_config_repo.py` 全部通过，含：插入两个版本后 `latest` 返回版本较高者且其 `effective_to` 为 NULL、旧版本 `effective_to` 非空、`by_version` 能取回任一历史版本
- 连续执行两次 `sync_from_seeds()`，第二次全部返回 `action="skip"`，`process_config` 行数不变
- `sync_from_seeds(dry_run=True)` 不产生数据库写入，`process_config` 行数不变
- `python -c` 断言从 `latest()` 取回的 `config_json` 可被 `ProcessConfig.model_validate` 成功还原，且 `content_hash` 与重新计算的哈希一致
- `ruff check`、`ruff format --check`、`mypy app` 三条命令返回 0
