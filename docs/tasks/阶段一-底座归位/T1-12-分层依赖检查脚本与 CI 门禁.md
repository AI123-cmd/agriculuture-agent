# T1-12 分层依赖检查脚本与 CI 门禁

## 这一项要做什么

建立 `scripts/check_layers.py`，用静态扫描强制四条硬性依赖规则：闸门不得依赖智能体与服务层、智能体不得直接导入 ORM、模型网关不得导入任何业务模块、任何模块不得直接使用 `db.query(Model)`。这个脚本不是"风格检查"，而是让安全闸门在物理上无法被绕过的执行手段——它接入 CI 后，违反即构建失败。

对应技术方案 15.2 的四条硬规则与产品说明书 9.1 的分层约束，也是 T1-06 整改结果的长期守卫：没有这道门禁，整改会在两次迭代内回到原样。

## 前置依赖

- 依赖 T1-05 已提供：`OwnedRepository` 作为 L4 规则的合法替代（白名单里的 `app/repositories/`）。
- 依赖 T1-06 已提供：42 处 `db.query` 已整改，脚本首次运行应为零违规。
- 依赖 T1-01 已提供：`ruff`、`mypy` 可运行，脚本与它们并列进 CI。
- 外部环境：CI 运行器需有 Python 3.11 与 Git 检出（脚本读 `git ls-files` 或遍历目录）。
- 需要人工确认：白名单条目。脚本必须支持显式白名单并写明理由，不允许 `# noqa` 跳过。

## 实现要点

### 一、check_layers.py 的完整骨架

```python
"""分层依赖检查：四条硬规则，违反即构建失败。

不提供行内豁免（# noqa）。确有例外必须在 check_layers_allowlist.toml 中
显式登记并写明理由，使例外可见、可审计、可定期清理。
"""
from __future__ import annotations

import ast
import sys
import tomllib
from dataclasses import dataclass
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]        # backend/
APP = ROOT / "app"
ALLOWLIST_FILE = ROOT / "scripts" / "check_layers_allowlist.toml"


# ---------- 规则定义 ----------

@dataclass(frozen=True)
class LayerRule:
    """一条 import 方向的禁令。"""

    name: str
    module_prefix: str            # 被检查的模块（如 "app.gate"）
    forbidden_prefixes: tuple[str, ...]
    reason: str                   # 违规时打印的理由（给执行者看，不只给机器看）


LAYER_RULES: tuple[LayerRule, ...] = (
    LayerRule(
        name="L1",
        module_prefix="app.gate",
        forbidden_prefixes=("app.agent", "app.services"),
        reason="闸门必须独立可测、可单独部署，这是生成与校验物理隔离的落地方式",
    ),
    LayerRule(
        name="L2",
        module_prefix="app.agent",
        forbidden_prefixes=("app.models",),
        reason="智能体需能被单测（喂假数据），耦合 ORM 就无法脱离数据库测试",
    ),
    LayerRule(
        name="L3",
        module_prefix="app.llm",
        forbidden_prefixes=("app.agent", "app.gate", "app.services", "app.broadcast"),
        reason="网关是通用组件，反向依赖会让它无法复用",
    ),
)

# L4：任何模块不得直接使用 db.query(Model)，必须走 repositories
DB_QUERY_PATTERN = "db.query("
DB_QUERY_ALLOWED_FILES = {
    "app/repositories/",          # 仓储是唯一合法入口
    "app/db/",                    # 基础设施
}


# ---------- 文件遍历与解析 ----------

def walk_py_files(prefix: str) -> list[Path]:
    """把模块前缀映射为目录，返回其中的全部 .py 文件。"""
    base = ROOT / prefix.replace(".", "/")
    if not base.exists():
        return []
    return sorted(p for p in base.rglob("*.py") if p.is_file())


def parse_imports(path: Path) -> list[str]:
    """用 AST 解析 import，返回被导入的完整模块名列表。

    不用正则匹配 import 行：正则会把字符串里的 "import app.agent" 误判，
    也会漏掉 `from app.agent.graph import main` 这类形式。
    """
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    imports: list[str] = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            imports.extend(alias.name for alias in node.names)
        elif isinstance(node, ast.ImportFrom):
            if node.module and node.level == 0:      # 只看绝对导入
                imports.append(node.module)
    return imports


def module_name_of(path: Path) -> str:
    """backend/app/agent/graph/main.py -> app.agent.graph.main"""
    rel = path.relative_to(ROOT).with_suffix("")
    parts = list(rel.parts)
    if parts[-1] == "__init__":
        parts.pop()
    return ".".join(parts)


# ---------- 白名单 ----------

@dataclass(frozen=True)
class AllowEntry:
    rule: str
    path: str            # 相对 backend/ 的路径或前缀
    reason: str


def load_allowlist() -> list[AllowEntry]:
    if not ALLOWLIST_FILE.exists():
        return []
    data = tomllib.loads(ALLOWLIST_FILE.read_text(encoding="utf-8"))
    return [
        AllowEntry(rule=e["rule"], path=e["path"], reason=e["reason"])
        for e in data.get("allow", [])
    ]


def is_allowed(allow: list[AllowEntry], rule: str, path: str) -> str | None:
    """命中白名单返回理由，否则 None。"""
    rel = path.replace("\\", "/")
    for entry in allow:
        if entry.rule == rule and rel.startswith(entry.path):
            return entry.reason
    return None


# ---------- 四条规则 ----------

def check_import_direction(allow: list[AllowEntry]) -> list[str]:
    """L1 / L2 / L3：import 方向禁令。"""
    violations: list[str] = []
    for rule in LAYER_RULES:
        for path in walk_py_files(rule.module_prefix):
            rel = str(path.relative_to(ROOT)).replace("\\", "/")
            if is_allowed(allow, rule.name, rel):
                continue
            for imp in parse_imports(path):
                if any(imp == f or imp.startswith(f + ".") for f in rule.forbidden_prefixes):
                    violations.append(
                        f"[{rule.name}] {rel}: 非法依赖 {imp} —— {rule.reason}"
                    )
    return violations


def check_db_query(allow: list[AllowEntry]) -> list[str]:
    """L4：任何文件不得直接使用 db.query(Model)。"""
    violations: list[str] = []
    for path in walk_py_files("app"):
        rel = str(path.relative_to(ROOT)).replace("\\", "/")
        if any(rel.startswith(p) for p in DB_QUERY_ALLOWED_FILES):
            continue
        if is_allowed(allow, "L4", rel):
            continue
        source = path.read_text(encoding="utf-8")
        tree = ast.parse(source, filename=str(path))
        for node in ast.walk(tree):
            # 只认真实的属性访问，不认注释与字符串
            if (
                isinstance(node, ast.Call)
                and isinstance(node.func, ast.Attribute)
                and node.func.attr == "query"
                and isinstance(node.func.value, ast.Name)
            ):
                violations.append(
                    f"[L4] {rel}:{node.lineno}: 禁止直接使用 {node.func.value.id}.query()，"
                    "请走 app/repositories/"
                )
    return violations


def check() -> list[str]:
    allow = load_allowlist()
    return check_import_direction(allow) + check_db_query(allow)


def main() -> int:
    violations = check()
    if violations:
        print(f"分层依赖检查未通过，共 {len(violations)} 项：", file=sys.stderr)
        for line in violations:
            print(f"  {line}", file=sys.stderr)
        print(
            "\n不得使用 # noqa 跳过。确有例外请在 "
            "backend/scripts/check_layers_allowlist.toml 登记并写明理由。",
            file=sys.stderr,
        )
        return 1
    print("分层依赖检查通过")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

### 二、为什么用 AST 而不是正则

现有技术方案 15.2 给的骨架是 `if "db.query(" in read(path)`，用子串匹配。这个写法有两个假阳性与一个假阴性：

假阳性一：注释里出现 `# db.query(User)` 会被判违规，导致执行者为了通过检查而删注释，反而降低了可读性。假阳性二：文档字符串里解释"不要用 db.query"的示例会被判违规。

假阴性：把 `session.query(Model)` 换成 `db.query(Model)` 之外的别名（`s = db; s.query(...)`）正则也能命中，但把 `db.query(\n  Model)` 跨行写时子串就在，正则也在；真正漏掉的是 `getattr(db, "query")(Model)` 这类。因此 AST 的意义不在防绕过（静态分析永远能绕过），而在于"只报真实的代码表达式"，使门禁结果可被信任。

`check_db_query` 里用 `isinstance(node.func.value, ast.Name)` 限定"直接名为 `db` 的对象"，会漏掉 `self.db.query(...)`。这是刻意的：`AsyncSession` 参数名在本项目的规范里固定为 `db`，若需要覆盖更多形态，把条件放宽为 `ast.Attribute` 即可，但会产生对 `obj.query` 的误报。本项选择按规范名匹配，并在脚本注释里写明"会话参数必须命名为 `db`"，这个约束写进 00-全局约定第四节的命名约定。

### 三、白名单文件

```toml
# 例外登记：每条必须有理由，且定期复核。
# 只登记"规则与现实的必要冲突"，不登记"懒得改"。

[[allow]]
rule = "L2"
path = "app/agent/nodes/"
reason = "第二步 T2-25 重排主图前，现有 nodes 通过 _db_from_config 取 Session；T2-25 完成后删除本条"

[[allow]]
rule = "L4"
path = "app/broadcast/snapshot.py"
reason = "第三步 T3-08 重建快照链路前保留现状查询；届时改走 repositories"
```

两条登记的共同特征：**都写明了失效条件**（哪个任务完成后必须删除）。这是白名单能否收敛的关键——没有失效条件的白名单会永久存在。建议在脚本里加一条元检查：白名单条目的 `reason` 必须含"T?"编号，否则报错：

```python
def check_allowlist_hygiene(allow: list[AllowEntry]) -> list[str]:
    bad = [e for e in allow if "T" not in e.reason or e.reason.count("T") != 1]
    return [f"[allowlist] {e.path} 的理由必须写明失效任务编号（如 T2-25）" for e in bad]
```

这是对技术方案的一处推断，方案只说"需在白名单里显式登记并写明理由"，未规定理由的格式。

### 四、CI 接入

`.github/workflows/ci.yml`：

```yaml
name: ci
on: [push, pull_request]
jobs:
  checks:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: timescale/timescaledb-ha:pg16
        env:
          POSTGRES_PASSWORD: huanong
        ports: ["5432:5432"]
        options: >-
          --health-cmd "pg_isready -U postgres" --health-interval 5s
          --health-timeout 5s --health-retries 10
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.11" }
      - run: pip install -r backend/requirements.txt -r backend/requirements-dev.txt
      - name: ruff
        run: ruff check backend/app && ruff format --check backend/app
      - name: mypy
        run: cd backend && mypy app
      - name: check-layers
        run: python backend/scripts/check_layers.py
      - name: migrations
        run: cd backend && alembic upgrade head && alembic downgrade -1 && alembic upgrade head
      - name: tests
        run: cd backend && pytest -q
```

`check-layers` 与 `ruff` / `mypy` 并列，顺序上放在 `mypy` 之后：分层违规往往同时导致类型错误，让更基础的门禁先报，可以避免同一问题被两个脚本重复报。

`migrations` 步骤执行 `upgrade head → downgrade -1 → upgrade head` 三段，落实技术方案 12.8 第三条"CI 中加检查：`alembic upgrade head` 与 `alembic downgrade -1` 均能成功执行"。

CI 用 `timescaledb-ha` 镜像而非官方 `postgres:16`，因为迁移依赖 `vector` 与 `timescaledb` 扩展（T1-03 的前置环境要求）。

### 六、脚本自身的测试

`tests/test_check_layers.py` 用临时目录构造违规样例，断言脚本能检出：

```python
def test_detects_gate_importing_agent(tmp_path, monkeypatch):
    fake = tmp_path / "app" / "gate"
    fake.mkdir(parents=True)
    (fake / "engine.py").write_text("from app.agent.graph import main\n")
    monkeypatch.setattr(check_layers, "ROOT", tmp_path)
    violations = check_layers.check()
    assert any("L1" in v for v in violations)
```

必须包含的四类正例（应检出）与四类反例（不应检出）：

- 正例：`app/gate/x.py` 写 `from app.services import s`；`app/agent/x.py` 写 `from app.models import User`；`app/llm/x.py` 写 `from app.gate import engine`；`app/services/x.py` 写 `db.query(User)`。
- 反例：注释 `# db.query(User)`；字符串 `"db.query(User)"`；`app/repositories/x.py` 里 `db.query`（白名单目录）；`app/gate/x.py` 导入 `app.process_config`（不在禁令内）。

反例里的第一条与第二条验证 AST 相对正则的优势；第三条验证目录白名单；第四条验证"禁止清单之外不禁"。

### 七、`app/gate/` 与 `app/llm/` 在本阶段是否已存在

第一步还没有建 `app/gate/`、`app/llm/`、`app/process_config/`，`walk_py_files` 对不存在的目录返回空列表，脚本不会报错。这一点要在脚本注释里说明，否则执行者会以为"检查没生效"。建议在 `main()` 里打印每个规则的扫描文件数，例如 `L1 app.gate: 0 个文件`，让"目录不存在"与"目录存在且合规"可区分。

### 八、与验收项的对应

- 直接对应技术方案 15.2 的四条硬规则，为产品说明书 9.1 分层与依赖约束提供自动化执行。
- 为产品说明书 9.2 第 1 条（归属）提供长期守卫：L4 规则让 `db.query` 在整改后无法回潮。
- 为技术方案 8.1"闸门必须独立可测"提供物理保证：L1 规则让闸门真的无法引用智能体与服务层。
- 为全局约定第三节"不允许 `# noqa` 跳过"提供实现：脚本不解析 `# noqa`，只认白名单文件。

## 验收标准

- `python backend/scripts/check_layers.py` 在已整改的代码库上返回 0，并打印 `分层依赖检查通过`。
- `python backend/scripts/check_layers.py` 的 stdout 或 stderr 打印每条规则的扫描文件数，且 `L1`、`L2`、`L3` 三行都可读（目录不存在时显示 0）。
- 在 `backend/app/services/` 任一文件里临时加入 `db.query(User)` 后运行脚本，返回 1，错误信息含 `[L4]`、文件名、行号与"请走 app/repositories/"。
- 在 `backend/app/services/` 任一文件里加入注释 `# db.query(User)` 后运行脚本，返回 0（AST 不误判）。
- 在 `backend/app/agent/` 任一文件里加入 `from app.models import User` 后运行脚本，返回 1 且错误信息含 `[L2]` 与原文理由。
- `pytest backend/tests/test_check_layers.py` 全部通过，含 4 条正例与 4 条反例，合计不少于 8 条断言。
- 白名单中每条 `[[allow]]` 的 `reason` 都含一个 `T` 开头的任务编号；构造一条不含编号的条目时脚本额外报 allowlist 卫生错误。
- `backend/scripts/check_layers_allowlist.toml` 的条目数不超过 5，且每条都有失效任务编号。
- CI 流水线 `checks` 作业在四个步骤（ruff / mypy / check-layers / migrations）全部返回 0 后才标记为成功。
- 在 CI 上人为制造一次 L4 违规并推送，流水线在 `check-layers` 步骤失败，且 `tests` 步骤不执行。
